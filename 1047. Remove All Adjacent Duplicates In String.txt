class Solution {
    public String removeDuplicates(String s) {
        
    Stack<Character> st = new Stack<>();
    StringBuffer sb = new StringBuffer();
    char[] ch = s.toCharArray();


    for(char c: ch)
    {
       if(st.size()>0)
       {
          if(st.peek()==c)
          {
            st.pop();
          }
          else{
            st.push(c);
          }

       }
       else{
         st.push(c);
       } 
    }

    while(st.size()>0)
    {
       sb.append(st.pop());
    }

   return sb.reverse().toString();
    }
}
